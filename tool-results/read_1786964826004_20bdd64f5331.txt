    55→function loadSettings(): Settings {
    56→  if (typeof window === "undefined") return { apiKey: "", stubMode: false };
    57→  try {
    58→    const raw = localStorage.getItem(LS_KEY);
    59→    if (raw) {
    60→      const parsed = JSON.parse(raw);
    61→      return { ...{ apiKey: "", stubMode: false }, ...parsed };
    62→    }
    63→  } catch {
    64→    /* ignore */
    65→  }
    66→  return { apiKey: "", stubMode: false };

   154→function renderMessageContent(text: string) {
   155→  // 1) Markdown-style links: [text](url)
   156→  // 2) Plain URLs: https://...
   157→  // Everything else stays as plain text
   158→  const parts: (string | { type: "link"; href: string; label: string })[] = [];
   159→  // Match markdown links first, then plain URLs; everything else is text
   160→  const pattern = /\[([^\]]+)\]\(([^)]+)\)|(https?:\/\/[^\s)<>]+)/g;
   161→  let lastIndex = 0;
   162→  let match: RegExpExecArray | null;
   163→
   164→  while ((match = pattern.exec(text)) !== null) {
   165→    if (match.index > lastIndex) {
   166→      parts.push(text.slice(lastIndex, match.index));
   167→    }
   168→    if (match[1] !== undefined) {
   169→      // Markdown link [label](href)
   170→      parts.push({ type: "link", href: match[2], label: match[1] });
   171→    } else {
   172→      // Plain URL
   173→      const url = match[0];
   174→      // Use the URL as label but truncate if very long
   175→      const label = url.length > 50 ? url.slice(0, 47) + "..." : url;
   176→      parts.push({ type: "link", href: url, label });
   177→    }
   178→    lastIndex = match.index + match[0].length;
   179→  }
   180→  if (lastIndex < text.length) {
   181→    parts.push(text.slice(lastIndex));
   182→  }
   183→
   184→  return parts.map((part, i) => {
   185→    if (typeof part === "string") {
   186→      // Preserve line breaks inside plain text segments
   187→      return part.split("\n").map((line, j, arr) => (
   188→        <span key={`${i}-${j}`}>
   189→          {line}
   190→          {j < arr.length - 1 && <br />}
   191→        </span>
   192→      ));
   193→    }
   194→    const isInternal = part.href.startsWith('/');
   195→    if (isInternal) {
   196→      return (
   197→        <Link
   198→          key={`l-${i}`}
   199→          href={part.href}
   200→          className="underline underline-offset-2 text-primary hover:text-primary/80 transition-colors"
   201→        >
   202→          {part.label}
   203→        </Link>
   204→      );
   205→    }
   206→    return (
   207→      <a
   208→        key={`l-${i}`}
   209→        href={part.href}
   210→        target="_blank"
   211→        rel="noopener noreferrer"
   212→        className="underline underline-offset-2 text-primary hover:text-primary/80 transition-colors"
   213→      >
   214→        {part.label}
   215→      </a>
   216→    );
   217→  });
   218→}

   233→async function callKimi(
   234→  apiKey: string,
   235→  systemPrompt: string,
   236→  userMessage: string
   237→): Promise<string> {
   238→  const url = "https://api.moonshot.ai/v1/chat/completions";
   239→
   240→  let res: Response;
   241→  try {
   242→    res = await fetch(url, {
   243→      method: "POST",
   244→      headers: {
   245→        "Content-Type": "application/json",
   246→        Authorization: `Bearer ${apiKey}`,
   247→      },
   248→      body: JSON.stringify({
   249→        model: KIMI_MODEL,
   250→        messages: [
   251→          { role: "system", content: systemPrompt },
   252→          { role: "user", content: userMessage },
   253→        ],
   254→      }),
   255→    });
   256→  } catch (err) {
   257→    if (err instanceof TypeError) {
   258→      throw new Error("Сетевая ошибка. Проверьте подключение к интернету.");
   259→    }
   260→    throw err;
   261→  }
   262→
   263→  if (!res.ok) {
   264→    let detail = `HTTP ${res.status}`;
   265→    try {
   266→      const body = await res.json();
   267→      detail = body?.error?.message || body?.error || detail;
   268→    } catch {
   269→      /* ignore */
   270→    }
   271→    if (res.status === 401)
   272→      detail = "Неверный API-ключ. Проверьте ключ на platform.moonshot.ai";
   273→    if (res.status === 402)
   274→      detail = "Недостаточно средств на аккаунте Kimi";
   275→    throw new Error(detail);
   276→  }
   277→
   278→  const data = await res.json();
   279→  const content = data.choices?.[0]?.message?.content;
   280→  if (!content) throw new Error("Пустой ответ от ИИ");
   281→  return content;
   282→}
