   162→   162→function renderMessageContent(text: string) {
   163→   163→  // 1) Markdown-style links: [text](url)
   164→   164→  // 2) Plain URLs: https://...
   165→   165→  // Everything else stays as plain text
   166→   166→  const parts: (string | { type: "link"; href: string; label: string })[] = [];
   167→   167→  // Match markdown links first, then plain URLs; everything else is text
   168→   168→  const pattern = /\[([^\]]+)\]\(([^)]+)\)|(https?:\/\/[^\s)<>]+)/g;
   169→   169→  let lastIndex = 0;
   170→   170→  let match: RegExpExecArray | null;
   171→   171→
   172→   172→  while ((match = pattern.exec(text)) !== null) {
   173→   173→    if (match.index > lastIndex) {
   174→   174→      parts.push(text.slice(lastIndex, match.index));
   175→   175→    }
   176→   176→    if (match[1] !== undefined) {
   177→   177→      // Markdown link [label](href)
   178→   178→      parts.push({ type: "link", href: match[2], label: match[1] });
   179→   179→    } else {
   180→   180→      // Plain URL
   181→   181→      const url = match[0];
   182→   182→      // Use the URL as label but truncate if very long
   183→   183→      const label = url.length > 50 ? url.slice(0, 47) + "..." : url;
   184→   184→      parts.push({ type: "link", href: url, label });
   185→   185→    }
   186→   186→    lastIndex = match.index + match[0].length;
   187→   187→  }
   188→   188→  if (lastIndex < text.length) {
   189→   189→    parts.push(text.slice(lastIndex));
   190→   190→  }
   191→   191→
   192→   192→  return parts.map((part, i) => {
   193→   193→    if (typeof part === "string") {
   194→   194→      // Preserve line breaks inside plain text segments
   195→   195→      return part.split("\n").map((line, j, arr) => (
   196→   196→        <span key={`${i}-${j}`}>
   197→   197→          {line}
   198→   198→          {j < arr.length - 1 && <br />}
   199→   199→        </span>
   200→   200→      ));
   201→   201→    }
   202→   202→    const isInternal = part.href.startsWith('/');
   203→   203→    if (isInternal) {
   204→   204→      return (
   205→   205→        <Link
   206→   206→          key={`l-${i}`}
   207→   207→          href={part.href}
   208→   208→          className="underline underline-offset-2 text-primary hover:text-primary/80 transition-colors"
   209→   209→        >
   210→   210→          {part.label}
   211→   211→        </Link>
   212→   212→      );
   213→   213→    }
   214→   214→    return (
   215→   215→      <a
   216→   216→        key={`l-${i}`}
   217→   217→        href={part.href}
   218→   218→        target="_blank"
   219→   219→        rel="noopener noreferrer"
   220→   220→        className="underline underline-offset-2 text-primary hover:text-primary/80 transition-colors"
   221→   221→      >
   222→   222→        {part.label}
   223→   223→      </a>
   224→   224→    );
   225→   225→  });
   226→   226→}

   248→   248→async function callKimiSingle(
   249→   249→  apiKey: string,
   250→   250→  model: string,
   251→   251→  systemPrompt: string,
   252→   252→  userMessage: string
   253→   253→): Promise<string> {
   254→   254→  const url = "https://api.moonshot.ai/v1/chat/completions";
   255→   255→
   256→   256→  let res: Response;
   257→   257→  try {
   258→   258→    res = await fetch(url, {
   259→   259→      method: "POST",
   260→   260→      headers: {
   261→   261→        "Content-Type": "application/json",
   262→   262→        Authorization: `Bearer ${apiKey}`,
   263→   263→      },
   264→   264→      body: JSON.stringify({
   265→   265→        model,
   266→   266→        messages: [
   267→   267→          { role: "system", content: systemPrompt },
   268→   268→          { role: "user", content: userMessage },
   269→   269→        ],
   270→   270→      }),
   271→   271→    });
   272→   272→  } catch (err) {
   273→   273→    if (err instanceof TypeError) {
   274→   274→      throw new Error("Сетевая ошибка. Проверьте подключение к интернету.");
   275→   275→    }
   276→   276→    throw err;
   277→   277→  }
   278→   278→
   279→   279→  if (!res.ok) {
   280→   280→    if (res.status === 429) throw new RateLimitError();
   281→   281→    let detail = `HTTP ${res.status}`;
   282→   282→    try {
   283→   283→      const body = await res.json();
   284→   284→      detail = body?.error?.message || body?.error || detail;
   285→   285→    } catch {
   286→   286→      /* ignore */
   287→   287→    }
   288→   288→    if (res.status === 401)
   289→   289→      detail = "Неверный API-ключ. Проверьте ключ на platform.moonshot.ai";
   290→   290→    if (res.status === 402)
   291→   291→      detail = "Недостаточно средств на аккаунте Kimi";
   292→   292→    throw new Error(detail);
   293→   293→  }
   294→   294→
   295→   295→  const data = await res.json();
   296→   296→  const content = data.choices?.[0]?.message?.content;
   297→   297→  if (!content) throw new Error("Пустой ответ от ИИ");
   298→   298→  return content;
   299→   299→}

   301→   301→async function callKimiWithFallback(
   302→   302→  apiKey: string,
   303→   303→  preferredModel: string,
   304→   304→  systemPrompt: string,
   305→   305→  userMessage: string
   306→   306→): Promise<string> {
   307→   307→  // Build ordered list: preferred first, then the rest
   308→   308→  const otherModels = KIMI_MODELS.filter((m) => m !== preferredModel);
   309→   309→  const modelsToTry = [preferredModel, ...otherModels];
   310→   310→  const errors: string[] = [];
   311→   311→
   312→   312→  for (const model of modelsToTry) {
   313→   313→    try {
   314→   314→      return await callKimiSingle(apiKey, model, systemPrompt, userMessage);
   315→   315→    } catch (err) {
   316→   316→      if (err instanceof RateLimitError) {
   317→   317→        errors.push(model);
   318→   318→        continue; // try next model
   319→   319→      }
   320→   320→      throw err; // non-429 errors should propagate immediately
   321→   321→    }
   322→   322→  }
   323→   323→
   324→   324→  // All models returned 429
   325→   325→  throw new Error(
   326→   326→    `Все модели перегружены (пробовались: ${errors.join(", ")}). Попробуйте позже.`
   327→   327→  );
   328→   328→}
