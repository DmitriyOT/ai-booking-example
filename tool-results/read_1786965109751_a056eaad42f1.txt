   162→function renderMessageContent(text: string) {
   163→  // 1) Markdown-style links: [text](url)
   164→  // 2) Plain URLs: https://...
   165→  // Everything else stays as plain text
   166→  const parts: (string | { type: "link"; href: string; label: string })[] = [];
   167→  // Match markdown links first, then plain URLs; everything else is text
   168→  const pattern = /\[([^\]]+)\]\(([^)]+)\)|(https?:\/\/[^\s)<>]+)/g;
   169→  let lastIndex = 0;
   170→  let match: RegExpExecArray | null;
   171→
   172→  while ((match = pattern.exec(text)) !== null) {
   173→    if (match.index > lastIndex) {
   174→      parts.push(text.slice(lastIndex, match.index));
   175→    }
   176→    if (match[1] !== undefined) {
   177→      // Markdown link [label](href)
   178→      parts.push({ type: "link", href: match[2], label: match[1] });
   179→    } else {
   180→      // Plain URL
   181→      const url = match[0];
   182→      // Use the URL as label but truncate if very long
   183→      const label = url.length > 50 ? url.slice(0, 47) + "..." : url;
   184→      parts.push({ type: "link", href: url, label });
   185→    }
   186→    lastIndex = match.index + match[0].length;
   187→  }
   188→  if (lastIndex < text.length) {
   189→    parts.push(text.slice(lastIndex));
   190→  }
   191→
   192→  return parts.map((part, i) => {
   193→    if (typeof part === "string") {
   194→      // Preserve line breaks inside plain text segments
   195→      return part.split("\n").map((line, j, arr) => (
   196→        <span key={`${i}-${j}`}>
   197→          {line}
   198→          {j < arr.length - 1 && <br />}
   199→        </span>
   200→      ));
   201→    }
   202→    const isInternal = part.href.startsWith('/');
   203→    if (isInternal) {
   204→      return (
   205→        <Link
   206→          key={`l-${i}`}
   207→          href={part.href}
   208→          className="underline underline-offset-2 text-primary hover:text-primary/80 transition-colors"
   209→        >
   210→          {part.label}
   211→        </Link>
   212→      );
   213→    }
   214→    return (
   215→      <a
   216→        key={`l-${i}`}
   217→        href={part.href}
   218→        target="_blank"
   219→        rel="noopener noreferrer"
   220→        className="underline underline-offset-2 text-primary hover:text-primary/80 transition-colors"
   221→      >
   222→        {part.label}
   223→      </a>
   224→    );
   225→  });
   226→}

   241→class RateLimitError extends Error {
   242→  constructor() {
   243→    super("RATE_LIMIT");
   244→    this.name = "RateLimitError";
   245→  }

   248→async function callKimiSingle(
   249→  apiKey: string,
   250→  model: string,
   251→  systemPrompt: string,
   252→  userMessage: string
   253→): Promise<string> {
   254→  const url = "https://api.moonshot.ai/v1/chat/completions";
   255→
   256→  let res: Response;
   257→  try {
   258→    res = await fetch(url, {
   259→      method: "POST",
   260→      headers: {
   261→        "Content-Type": "application/json",
   262→        Authorization: `Bearer ${apiKey}`,
   263→      },
   264→      body: JSON.stringify({
   265→        model,
   266→        messages: [
   267→          { role: "system", content: systemPrompt },
   268→          { role: "user", content: userMessage },
   269→        ],
   270→      }),
   271→    });
   272→  } catch (err) {
   273→    if (err instanceof TypeError) {
   274→      throw new Error("Сетевая ошибка. Проверьте подключение к интернету.");
   275→    }
   276→    throw err;
   277→  }
   278→
   279→  if (!res.ok) {
   280→    if (res.status === 429) throw new RateLimitError();
   281→    let detail = `HTTP ${res.status}`;
   282→    try {
   283→      const body = await res.json();
   284→      detail = body?.error?.message || body?.error || detail;
   285→    } catch {
   286→      /* ignore */
   287→    }
   288→    if (res.status === 401)
   289→      detail = "Неверный API-ключ. Проверьте ключ на platform.moonshot.ai";
   290→    if (res.status === 402)
   291→      detail = "Недостаточно средств на аккаунте Kimi";
   292→    throw new Error(detail);
   293→  }
   294→
   295→  const data = await res.json();
   296→  const content = data.choices?.[0]?.message?.content;
   297→  if (!content) throw new Error("Пустой ответ от ИИ");
   298→  return content;
   299→}

   301→async function callKimiWithFallback(
   302→  apiKey: string,
   303→  preferredModel: string,
   304→  systemPrompt: string,
   305→  userMessage: string
   306→): Promise<string> {
   307→  // Build ordered list: preferred first, then the rest
   308→  const otherModels = KIMI_MODELS.filter((m) => m !== preferredModel);
   309→  const modelsToTry = [preferredModel, ...otherModels];
   310→  const errors: string[] = [];
   311→
   312→  for (const model of modelsToTry) {
   313→    try {
   314→      return await callKimiSingle(apiKey, model, systemPrompt, userMessage);
   315→    } catch (err) {
   316→      if (err instanceof RateLimitError) {
   317→        errors.push(model);
   318→        continue; // try next model
   319→      }
   320→      throw err; // non-429 errors should propagate immediately
   321→    }
   322→  }
   323→
   324→  // All models returned 429
   325→  throw new Error(
   326→    `Все модели перегружены (пробовались: ${errors.join(", ")}). Попробуйте позже.`
   327→  );
   328→}
